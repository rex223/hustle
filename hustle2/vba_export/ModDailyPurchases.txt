
Public Sub SaveDailyPurchases()
    Dim wsEntry As Worksheet
    Dim wsDB As Worksheet
    Dim tblEntry As ListObject
    Dim tblDB As ListObject
    Dim dateCell As Range
    
    On Error Resume Next
    Set wsEntry = ThisWorkbook.Sheets("PURCHASE SHEET")
    Set wsDB = ThisWorkbook.Sheets("DATABASE_PURCHASE")
    Set tblEntry = wsEntry.ListObjects("EntryFormTable")
    Set tblDB = wsDB.ListObjects("DailyPurchaseLedger")
    Set dateCell = wsEntry.Range("C1")
    On Error GoTo 0
    
    If wsEntry Is Nothing Or wsDB Is Nothing Then Exit Sub
    If Not IsDate(dateCell.Value) Then Exit Sub
    
    Dim saveDate As Date
    saveDate = CDate(dateCell.Value)
    
    Application.ScreenUpdating = False
    Application.EnableEvents = False
    
    Dim i As Long, k As Long
    
    ' Loop through every item on the Purchase Sheet
    For i = 1 To tblEntry.DataBodyRange.Rows.Count
        Dim qtyKg As String, qtyPcs As String, rate As String, trsp As String, total As String
        qtyKg = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 3).Value & ""))
        qtyPcs = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 4).Value & ""))
        rate = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 5).Value & ""))
        trsp = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 6).Value & ""))
        total = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 7).Value & ""))
        
        Dim entryGroupName As String, entryItemName As String
        entryGroupName = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 1).Value & ""))
        entryItemName = Trim(CStr(tblEntry.DataBodyRange.Cells(i, 2).Value & ""))
        
        Dim hasData As Boolean
        hasData = False
        ' Check if there is any data (even text like "9PC")
        If qtyKg <> "" Or qtyPcs <> "" Or rate <> "" Or trsp <> "" Or total <> "" Then
            If Val(qtyKg) > 0 Or Val(qtyPcs) > 0 Or Val(rate) > 0 Or Val(trsp) > 0 Or Val(total) > 0 Then
                hasData = True
            End If
        End If
        
        ' Only proceed if this row actually has data entered in the form
        If hasData Then
            Dim foundRow As Long
            foundRow = 0
            
            ' Look for this exact item on this exact date in the Database
            If Not tblDB.DataBodyRange Is Nothing Then
                For k = 1 To tblDB.DataBodyRange.Rows.Count
                    If IsDate(tblDB.DataBodyRange.Cells(k, 1).Value) Then
                        If CDate(tblDB.DataBodyRange.Cells(k, 1).Value) = saveDate Then
                            If UCase(Trim(tblDB.DataBodyRange.Cells(k, 3).Value)) = UCase(entryItemName) Then
                                foundRow = k
                                Exit For
                            End If
                        End If
                    End If
                Next k
            End If
            
            If foundRow > 0 Then
                ' IT EXISTS -> Safely update the row in place. NO DELETIONS. NO SHIFTING.
                tblDB.DataBodyRange.Cells(foundRow, 4).Value = tblEntry.DataBodyRange.Cells(i, 3).Value
                tblDB.DataBodyRange.Cells(foundRow, 5).Value = tblEntry.DataBodyRange.Cells(i, 4).Value
                tblDB.DataBodyRange.Cells(foundRow, 6).Value = tblEntry.DataBodyRange.Cells(i, 5).Value
                tblDB.DataBodyRange.Cells(foundRow, 7).Value = tblEntry.DataBodyRange.Cells(i, 6).Value
                tblDB.DataBodyRange.Cells(foundRow, 8).Value = tblEntry.DataBodyRange.Cells(i, 7).Value
            Else
                ' IT DOES NOT EXIST -> Safely add it to the bottom
                Dim r As ListRow
                Set r = tblDB.ListRows.Add
                r.Range(1, 1).Value = saveDate
                r.Range(1, 2).Value = entryGroupName
                r.Range(1, 3).Value = entryItemName
                r.Range(1, 4).Value = tblEntry.DataBodyRange.Cells(i, 3).Value
                r.Range(1, 5).Value = tblEntry.DataBodyRange.Cells(i, 4).Value
                r.Range(1, 6).Value = tblEntry.DataBodyRange.Cells(i, 5).Value
                r.Range(1, 7).Value = tblEntry.DataBodyRange.Cells(i, 6).Value
                r.Range(1, 8).Value = tblEntry.DataBodyRange.Cells(i, 7).Value
            End If
        End If
    Next i
    
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    
    MsgBox "Update complete! Only changed rows were updated. Database structure is completely untouched.", vbInformation, "Perfect Update"
End Sub
