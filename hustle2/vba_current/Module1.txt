Sub SaveExpenditureData()
    Dim wsToday As Worksheet
    Dim wsDB As Worksheet
    Set wsToday = ThisWorkbook.Sheets("todays_expenditure")
    Set wsDB = ThisWorkbook.Sheets("database_expenditure")
    
    Dim monthMarker As String
    monthMarker = UCase(Trim(wsToday.Range("A2").Value))
    
    If Not monthMarker Like "*_20##" Then
        MsgBox "Please enter a valid month marker (e.g., JULY_2026) in cell A2!", vbExclamation
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    
    ' --- FIX 1: IGNORE THE SUMMARY TABLE (Stop EXACTLY at TOTAL row) ---
    Dim lastRowToday As Long
    lastRowToday = wsToday.Cells(wsToday.Rows.Count, "B").End(xlUp).Row
    
    Dim r As Long
    For r = 1 To lastRowToday
        If UCase(Trim(wsToday.Cells(r, 1).Value)) = "TOTAL" Or UCase(Trim(wsToday.Cells(r, 1).Value)) = "GRAND TOTAL" Then
            lastRowToday = r ' Stop EXACTLY at the TOTAL row
            Exit For
        End If
    Next r
    
    If lastRowToday < 2 Then lastRowToday = 2
    
    Dim i As Long
    Dim foundStart As Long, foundEnd As Long
    foundStart = 0
    
    Dim lastRowDB As Long
    lastRowDB = wsDB.Cells(wsDB.Rows.Count, "B").End(xlUp).Row
    If lastRowDB < 2 Then lastRowDB = 2
    
    ' 1. Find if this month already exists in the Database
    For i = 2 To lastRowDB
        If UCase(Trim(wsDB.Cells(i, 1).Value)) = monthMarker Then
            foundStart = i
            Exit For
        End If
    Next i
    
    ' 2. If it exists, delete the old block so we can replace it with the updated one
    If foundStart > 0 Then
        foundEnd = foundStart
        For i = foundStart + 1 To lastRowDB
            If UCase(Trim(wsDB.Cells(i, 1).Value)) Like "*_20##" Then
                foundEnd = i - 1
                Exit For
            End If
        Next i
        If foundEnd = foundStart Then foundEnd = lastRowDB
        
        wsDB.Rows(foundStart & ":" & foundEnd).Delete
    End If
    
    ' 3. Insert the new block exactly where the old one was (or at the bottom if new)
    Dim insertRow As Long
    If foundStart > 0 Then
        insertRow = foundStart
    Else
        insertRow = wsDB.Cells(wsDB.Rows.Count, "B").End(xlUp).Row + 1
        If insertRow < 2 Then insertRow = 2
    End If
    
    ' --- FIX 2: PREVENT BUTTONS FROM COPYING ---
    Application.CopyObjectsWithCells = False
    wsToday.Rows("2:" & lastRowToday).Copy
    wsDB.Rows(insertRow).Insert Shift:=xlDown
    Application.CopyObjectsWithCells = True
    Application.CutCopyMode = False
    
    Application.ScreenUpdating = True
    MsgBox monthMarker & " data saved successfully!", vbInformation, "Saved"
End Sub

Sub LoadExpenditureData()
    Dim wsToday As Worksheet
    Dim wsDB As Worksheet
    Set wsToday = ThisWorkbook.Sheets("todays_expenditure")
    Set wsDB = ThisWorkbook.Sheets("database_expenditure")
    
    Dim monthMarker As String
    monthMarker = InputBox("Enter the month to load exactly as saved (e.g., JULY_2026):", "Load Data")
    If monthMarker = "" Then Exit Sub
    monthMarker = UCase(Trim(monthMarker))
    
    Application.ScreenUpdating = False
    
    Dim lastRowDB As Long
    lastRowDB = wsDB.Cells(wsDB.Rows.Count, "B").End(xlUp).Row
    
    Dim i As Long
    Dim foundStart As Long, foundEnd As Long
    foundStart = 0
    
    For i = 2 To lastRowDB
        If UCase(Trim(wsDB.Cells(i, 1).Value)) = monthMarker Then
            foundStart = i
            Exit For
        End If
    Next i
    
    If foundStart = 0 Then
        MsgBox "Data for " & monthMarker & " was not found in the database.", vbExclamation
        Application.ScreenUpdating = True
        Exit Sub
    End If
    
    foundEnd = foundStart
    For i = foundStart + 1 To lastRowDB
        If UCase(Trim(wsDB.Cells(i, 1).Value)) Like "*_20##" Then
            foundEnd = i - 1
            Exit For
        End If
    Next i
    If foundEnd = foundStart Then foundEnd = lastRowDB
    
    ' SAFELY CLEAR THE OLD DATA WITHOUT DELETING ROWS OR BUTTONS!
    Dim lastRowToday As Long
    lastRowToday = wsToday.Cells(wsToday.Rows.Count, "B").End(xlUp).Row
    If lastRowToday >= 2 Then
        ' .Clear removes data and formatting, but leaves shapes/buttons completely safe!
        wsToday.Rows("2:" & (lastRowToday + 5)).Clear
    End If
    
    ' Copy data from DB (With CopyObjectsWithCells turned OFF so rogue DB shapes don't come back)
    Application.CopyObjectsWithCells = False
    wsDB.Rows(foundStart & ":" & foundEnd).Copy
    wsToday.Range("A2").PasteSpecial Paste:=xlPasteAll
    Application.CutCopyMode = False
    Application.CopyObjectsWithCells = True
    
    ' Force the summary table to recalculate since we pasted new data
    On Error Resume Next
    Call ThisWorkbook.UpdateExpenditureSummary
    On Error GoTo 0
    
    Application.ScreenUpdating = True
    MsgBox monthMarker & " loaded successfully!", vbInformation, "Loaded"
End Sub

Sub StartNewMonth()
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("todays_expenditure")
    
    Dim newMonth As String
    newMonth = InputBox("Enter the marker for the NEW month (e.g., AUGUST_2026):", "New Month")
    If newMonth = "" Then Exit Sub
    If Not UCase(Trim(newMonth)) Like "*_20##" Then
        MsgBox "Marker must end with an underscore and year (e.g., _2026)", vbCritical
        Exit Sub
    End If
    
    Application.ScreenUpdating = False
    ws.Range("A2").Value = UCase(Trim(newMonth))
    
    Dim lastRow As Long
    lastRow = 35 ' Default TOTAL row
    Dim r As Long
    For r = 1 To 100
        If UCase(Trim(ws.Cells(r, 1).Value)) = "TOTAL" Or UCase(Trim(ws.Cells(r, 1).Value)) = "GRAND TOTAL" Then
            lastRow = r
            Exit For
        End If
    Next r
    
    If lastRow > 3 Then
        Dim c As Long, header As String
        On Error Resume Next
        For c = 4 To 50
            header = UCase(Trim(ws.Cells(1, c).Value))
            If header = "DATE" Or header = "AMOUNT" Or header = "PAID" Or header = "PAID STATUS" Or header = "ADV" Then
                ws.Range(ws.Cells(3, c), ws.Cells(lastRow - 1, c)).ClearContents
            End If
        Next c
        On Error GoTo 0
    End If
    
    ' Re-assert TOTAL formulas on row lastRow
    ws.Cells(lastRow, 2).Formula = "=SUM(B3:B" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 8).Formula = "=SUM(H3:H" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 12).Formula = "=SUM(L3:L" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 16).Formula = "=SUM(P3:P" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 20).Formula = "=SUM(T3:T" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 24).Formula = "=SUM(X3:X" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 28).Formula = "=SUM(AB3:AB" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 32).Formula = "=SUM(AF3:AF" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 36).Formula = "=SUM(AJ3:AJ" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 41).Formula = "=SUM(AO3:AO" & (lastRow - 1) & ")"
    ws.Cells(lastRow, 44).Formula = "=SUM(AR3:AR" & (lastRow - 1) & ")"
    
    Application.ScreenUpdating = True
    MsgBox "Sheet is cleared and ready for " & newMonth & "! The fixed salaries/details and TOTAL formulas have been preserved.", vbInformation, "Ready"
End Sub

' =========================================================================
' RENT MISC - SAVE TO DATABASE_RENT_MISC
' =========================================================================
Sub SaveRentMiscData()
    Dim wsTRM As Worksheet, wsDBRM As Worksheet
    On Error Resume Next
    Set wsTRM = ThisWorkbook.Sheets("TODAYS_RENT_MISC")
    Set wsDBRM = ThisWorkbook.Sheets("DATABASE_RENT_MISC")
    On Error GoTo 0
    
    If wsTRM Is Nothing Then
        MsgBox "TODAYS_RENT_MISC sheet not found!", vbExclamation
        Exit Sub
    End If
    If wsDBRM Is Nothing Then
        MsgBox "DATABASE_RENT_MISC sheet not found!", vbExclamation
        Exit Sub
    End If
    
    Dim monthMarker As String
    monthMarker = UCase(Trim(wsTRM.Cells(1, 1).Value))
    If InStr(monthMarker, ChrW(8212)) > 0 Then
        monthMarker = Trim(Mid(monthMarker, InStr(monthMarker, ChrW(8212)) + 1))
    ElseIf InStr(monthMarker, "-") > 0 Then
        monthMarker = Trim(Mid(monthMarker, InStr(monthMarker, "-") + 1))
    End If
    
    If monthMarker = "" Or InStr(monthMarker, "_20") = 0 Then
        monthMarker = InputBox("Enter the month marker (e.g., OCTOBER_2026):", "Save Misc Data")
        If monthMarker = "" Then Exit Sub
        monthMarker = UCase(Trim(monthMarker))
    End If
    
    Application.ScreenUpdating = False
    
    Dim lastDBRow As Long
    lastDBRow = wsDBRM.Cells(wsDBRM.Rows.Count, "A").End(xlUp).Row
    
    Dim foundStart As Long, foundEnd As Long, i As Long
    foundStart = 0
    For i = 2 To lastDBRow
        If InStr(UCase(Trim(wsDBRM.Cells(i, 1).Value)), monthMarker) > 0 Then
            foundStart = i
            Exit For
        End If
    Next i
    
    If foundStart > 0 Then
        foundEnd = foundStart
        For i = foundStart + 1 To lastDBRow
            Dim nextVal As String
            nextVal = UCase(Trim(wsDBRM.Cells(i, 1).Value))
            If InStr(nextVal, "_20") > 0 And InStr(nextVal, "DATABASE") = 0 And i > foundStart + 2 Then
                foundEnd = i - 1
                Exit For
            End If
        Next i
        If foundEnd = foundStart Then foundEnd = lastDBRow
        wsDBRM.Rows(foundStart & ":" & foundEnd).Delete
    End If
    
    Dim totRow As Long, dataStart As Long
    dataStart = 4
    totRow = 0
    For i = 4 To 500
        If UCase(Trim(wsTRM.Cells(i, 1).Value)) = "TOTAL" Or UCase(Trim(wsTRM.Cells(i, 7).Value)) = "TOTAL" Then
            totRow = i
            Exit For
        End If
    Next i
    If totRow = 0 Then
        MsgBox "Could not find TOTAL row in TODAYS_RENT_MISC.", vbExclamation
        Application.ScreenUpdating = True
        Exit Sub
    End If
    
    Dim insertRow As Long
    insertRow = wsDBRM.Cells(wsDBRM.Rows.Count, "A").End(xlUp).Row + 2
    If insertRow < 4 Then insertRow = 4
    
    wsDBRM.Rows(insertRow).Insert Shift:=xlDown
    wsDBRM.Cells(insertRow, 1).Value = monthMarker
    With wsDBRM.Range(wsDBRM.Cells(insertRow, 1), wsDBRM.Cells(insertRow, 10))
        .Interior.Color = RGB(44, 62, 80)
        .Font.Color = RGB(255, 255, 255)
        .Font.Bold = True
    End With
    
    Application.CopyObjectsWithCells = False
    wsTRM.Rows(dataStart & ":" & totRow).Copy
    wsDBRM.Rows(insertRow + 1).Insert Shift:=xlDown
    Application.CutCopyMode = False
    Application.CopyObjectsWithCells = True
    
    Application.ScreenUpdating = True
    MsgBox monthMarker & " saved to DATABASE_RENT_MISC!", vbInformation, "Saved"
End Sub

' =========================================================================
' RENT MISC - LOAD FROM DATABASE_RENT_MISC
' =========================================================================
Sub LoadRentMiscData()
    Dim wsTRM As Worksheet, wsDBRM As Worksheet
    On Error Resume Next
    Set wsTRM = ThisWorkbook.Sheets("TODAYS_RENT_MISC")
    Set wsDBRM = ThisWorkbook.Sheets("DATABASE_RENT_MISC")
    On Error GoTo 0
    
    If wsTRM Is Nothing Or wsDBRM Is Nothing Then
        MsgBox "TODAYS_RENT_MISC or DATABASE_RENT_MISC not found!", vbExclamation
        Exit Sub
    End If
    
    Dim monthMarker As String
    monthMarker = InputBox("Enter the month to load (e.g., SEPTEMBER_2026):", "Load Rent Misc")
    If monthMarker = "" Then Exit Sub
    monthMarker = UCase(Trim(monthMarker))
    
    Application.ScreenUpdating = False
    
    Dim lastDBRow As Long
    lastDBRow = wsDBRM.Cells(wsDBRM.Rows.Count, "A").End(xlUp).Row
    
    Dim foundStart As Long, foundEnd As Long, i As Long
    foundStart = 0
    For i = 2 To lastDBRow
        If InStr(UCase(Trim(wsDBRM.Cells(i, 1).Value)), monthMarker) > 0 Then
            foundStart = i
            Exit For
        End If
    Next i
    
    If foundStart = 0 Then
        MsgBox "Data for " & monthMarker & " not found in DATABASE_RENT_MISC.", vbExclamation
        Application.ScreenUpdating = True
        Exit Sub
    End If
    
    foundEnd = foundStart
    For i = foundStart + 1 To lastDBRow
        Dim nv As String
        nv = UCase(Trim(wsDBRM.Cells(i, 1).Value))
        If InStr(nv, "_20") > 0 And InStr(nv, "DATABASE") = 0 And i > foundStart + 3 Then
            foundEnd = i - 1
            Exit For
        End If
    Next i
    If foundEnd = foundStart Then foundEnd = lastDBRow
    
    Dim lastTRMRow As Long
    lastTRMRow = wsTRM.Cells(wsTRM.Rows.Count, "A").End(xlUp).Row
    If lastTRMRow >= 4 Then
        wsTRM.Rows("4:" & (lastTRMRow + 5)).Clear
    End If
    
    Application.CopyObjectsWithCells = False
    wsDBRM.Rows((foundStart + 1) & ":" & foundEnd).Copy
    wsTRM.Range("A4").PasteSpecial Paste:=xlPasteAll
    Application.CutCopyMode = False
    Application.CopyObjectsWithCells = True
    
    wsTRM.Cells(1, 1).Value = "RENT PERSONS MISCELLANEOUS  " & ChrW(8212) & "  " & monthMarker
    
    Application.ScreenUpdating = True
    MsgBox monthMarker & " loaded into TODAYS_RENT_MISC!", vbInformation, "Loaded"
End Sub

' =========================================================================
' RENT MISC - START NEW MONTH
' =========================================================================
Sub StartNewRentMiscMonth()
    Dim wsTRM As Worksheet
    On Error Resume Next
    Set wsTRM = ThisWorkbook.Sheets("TODAYS_RENT_MISC")
    On Error GoTo 0
    
    If wsTRM Is Nothing Then
        MsgBox "TODAYS_RENT_MISC not found!", vbExclamation
        Exit Sub
    End If
    
    Dim confirm As Integer
    confirm = MsgBox("Save current data to DATABASE_RENT_MISC and clear for new month?", vbYesNo + vbQuestion, "Start New Month")
    If confirm <> vbYes Then Exit Sub
    
    Call SaveRentMiscData
    
    Dim newMonth As String
    newMonth = InputBox("Enter the new month marker (e.g., NOVEMBER_2026):", "New Month")
    If newMonth = "" Then Exit Sub
    newMonth = UCase(Trim(newMonth))
    
    Application.ScreenUpdating = False
    
    Dim totRow As Long, i As Long
    totRow = 0
    For i = 4 To 500
        If UCase(Trim(wsTRM.Cells(i, 1).Value)) = "TOTAL" Or UCase(Trim(wsTRM.Cells(i, 7).Value)) = "TOTAL" Then
            totRow = i
            Exit For
        End If
    Next i
    
    If totRow > 4 Then
        wsTRM.Rows("4:" & (totRow - 1)).Delete
    End If
    
    wsTRM.Cells(1, 1).Value = "RENT PERSONS MISCELLANEOUS  " & ChrW(8212) & "  " & newMonth
    
    ' Re-apply formulas to TOTAL row
    totRow = 0
    For i = 4 To 500
        If UCase(Trim(wsTRM.Cells(i, 1).Value)) = "TOTAL" Or UCase(Trim(wsTRM.Cells(i, 7).Value)) = "TOTAL" Then
            totRow = i
            Exit For
        End If
    Next i
    If totRow > 0 Then
        wsTRM.Cells(totRow, 4).Formula = "=SUM(D4:D" & (totRow - 1) & ")"
        wsTRM.Cells(totRow, 10).Formula = "=SUM(J4:J" & (totRow - 1) & ")"
        wsTRM.Cells(totRow + 1, 4).Formula = "=D" & totRow & "-J" & totRow
    End If
    
    Application.ScreenUpdating = True
    MsgBox "TODAYS_RENT_MISC ready for " & newMonth & "!", vbInformation, "Ready"
End Sub
